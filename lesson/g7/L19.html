ប្រវត្តិនៃភាពស៊ីមេទ្រី និងភាពឆ្លុះ

«ភាពស៊ីមេទ្រី និងភាពឆ្លុះ» (Symmetry and Reflection) គឺជាគោលការណ៍គ្រឹះនៃសោភ័ណភាព និងតុល្យភាពនៅក្នុងសកលលោក។ នៅក្នុងធម្មជាតិ យើងអាចសង្កេតឃើញភាពឆ្លុះស្ទើរគ្រប់ទីកន្លែង ចាប់តាំងពីរូបរាងស្លាបមេអំបៅ ផ្កាឈូក ស្លឹកឈើ ក្រឡាព្រិលទឹកកក រហូតដល់ទម្រង់រាងកាយមនុស្ស និងសត្វ។ តាំងពីសម័យបុរាណ ជនជាតិក្រិក និងរ៉ូមបានចាត់ទុកភាពស៊ីមេទ្រីថាជាមូលដ្ឋានគ្រឹះនៃសិល្បៈដ៏ល្អឥតខ្ចោះ។ ក្នុងគណិតវិទ្យា ភាពឆ្លុះគឺជាបំប្លែងធរណីមាត្រមួយប្រភេទ (Isometry) ដែលរក្សាទំហំ ប្រវែង មុំ និងក្រឡាផ្ទៃរបស់រូបធរណីមាត្រឱ្យនៅដដែល ដោយគ្រាន់តែផ្លាស់ប្ដូរទីតាំង ឬទិសដៅប៉ុណ្ណោះ។

នៅក្នុងប្រវត្តិសាស្ត្រ និងសិល្បៈស្ថាបត្យកម្មខ្មែរ ភាពស៊ីមេទ្រីត្រូវបានបុព្វបុរសខ្មែរយកមកប្រើប្រាស់ប្រកបដោយភាពច្នៃប្រឌិត និងទេពកោសល្យកំពូល។ ប្លង់មេនៃប្រាសាទអង្គរវត្ត ប្រាសាទបាគង និងប្រាសាទបាយ័ន សុទ្ធសឹងតែត្រូវបានរៀបចំឡើងតាមអ័ក្សស៊ីមេទ្រីដ៏អស្ចារ្យពីកើតទៅលិច និងពីជើងទៅត្បូង។ លើសពីនេះ ក្បាច់ចម្លាក់ផ្ដែរ ហោជាង ស្លាបនាគ និងចម្លាក់ទេវតាទាញព្រ័ត្រជាមួយយក្សនៅលើស្ពាននាគ ឆ្លុះបញ្ចាំងពីតុល្យភាពស៊ីមេទ្រីយ៉ាងល្អប្រណីត។ ក្នុងវិទ្យាសាស្ត្រ និងបច្ចេកវិទ្យាទំនើប ភាពឆ្លុះជាគោលការណ៍ចម្បងក្នុងការរចនាយានយន្ត យន្តហោះ កញ្ចក់អុបទិក ក្រាហ្វិកកុំព្យូទ័រ និងការសិក្សាអំពីរចនាសម្ព័ន្ធគ្រីស្តាល់ និងម៉ូលេគុល DNA។

សញ្ញាណទូទៅនៃភាពឆ្លុះ និងបំប្លែងធរណីមាត្រ

នៅក្នុងធរណីមាត្រប្លង់ គេបែងចែកភាពឆ្លុះជាបីទម្រង់សំខាន់ៗ ៖

  • ភាពឆ្លុះធៀបនឹងចំណុច (Central Symmetry) ៖ ជាបំប្លែងដែលបង្វែរគ្រប់ចំណុចកាត់តាមចំណុចនឹងមួយហៅថា ផ្ចិតឆ្លុះ (\(O\)) ដោយផ្ចិតឆ្លុះជាចំណុចកណ្ដាលនៃអង្កត់ភ្ជាប់រវាងចំណុចដើម និងចំណុចរូបភាព។

  • ភាពឆ្លុះធៀបនឹងបន្ទាត់ (Axial Reflection) ៖ ជាបំប្លែងឆ្លុះធៀបនឹងបន្ទាត់នឹងមួយហៅថា អ័ក្សឆ្លុះ (\(d\)) ដោយបន្ទាត់នោះជាមេដ្យាទ័រនៃអង្កត់ភ្ជាប់រវាងចំណុចដើម និងចំណុចរូបភាព។

  • បំប្លែងកិល (Translation) ៖ ជាការរំកិលរូបទាំងមូលទៅតាមទិស ទិសដៅ និងប្រវែងនៃវ៉ិចទ័រមួយកំណត់។

  • លក្ខណៈរួម ៖ រាល់បំប្លែងឆ្លុះ និងបំប្លែងកិល រក្សាប្រវែងអង្កត់ រង្វាស់មុំ ភាពត្រង់ជួរនៃចំណុច និងក្រឡាផ្ទៃ (រូបភាពប៉ុនរូបដើមជានិច្ច)។

ភាពឆ្លុះធៀបនឹងចំណុច (ផ្ចិតឆ្លុះ)

ចំណុចឆ្លុះធៀបនឹងចំណុចមួយ

  • និយមន័យ ៖ ចំណុច \(M'\) ឆ្លុះនឹងចំណុច \(M\) ធៀបនឹងចំណុច \(O\) កាលណា \(O\) ជាចំណុចកណ្ដាលនៃអង្កត់ \([MM']\)។

  • ផ្ចិតឆ្លុះ ៖ ចំណុច \(O\) ហៅថា ផ្ចិតឆ្លុះ នៃភាពឆ្លុះ។

  • លក្ខណៈ ៖ \(M, O, M'\) រត់ត្រង់ជួរគ្នា និង \(OM = OM'\)។

  • ចំណុចនៅនឹង ៖ ចំណុចឆ្លុះនៃផ្ចិតឆ្លុះ \(O\) ធៀបនឹង \(O\) គឺខ្លួនវាផ្ទាល់ (\(O' \equiv O\))។

ឧទាហរណ៍

នៅលើបន្ទាត់មួយ គេមានចំណុច \(A\) និង \(O\) ដោយ \(OA = 4\text{ cm}\)។
ក. សង់ចំណុច \(A'\) ឆ្លុះនឹង \(A\) ធៀបនឹង \(O\)។ គណនាប្រវែង \(AA'\)។
ខ. បើ \(I\) ជាចំណុចកណ្ដាលនៃ \([OA]\) តើចំណុចឆ្លុះ \(I'\) នៃ \(I\) ធៀបនឹង \(O\) នៅត្រង់ណា? គណនា \(OI'\)។

ដំណោះស្រាយ

ដំណោះស្រាយលម្អិត៖
 ដោយ \(A'\) ឆ្លុះនឹង \(A\) ធៀបនឹង \(O\) នោះ \(O\) ជាចំណុចកណ្ដាលនៃ \([AA'] \implies OA' = OA = 4\text{ cm}\)។
ប្រវែង \(AA' = OA + OA' = 4 + 4 = 8\text{ cm}\)។ ដូចនេះ \(\boxed{AA' = 8\text{ cm}}\)
 ដោយ \(I\) ជាចំណុចកណ្ដាលនៃ \([OA]\) នោះ \(OI = \frac{OA}{2} = 2\text{ cm}\)។ ចំណុចឆ្លុះ \(I'\) ធៀបនឹង \(O\) ស្ថិតនៅលើកាំរស្មីផ្ទុយនឹង \([OI)\) ដោយ \(OI' = OI = 2\text{ cm}\)។ ដូចនេះ \(\boxed{OI' = 2\text{ cm}}\)

១. គេមានអង្កត់ \([AB]\) ប្រវែង \(6\text{ cm}\) និងចំណុច \(O\) មួយនៅក្រៅអង្កត់នោះ។ ចូរសង់ចំណុច \(A'\) និង \(B'\) ឆ្លុះនឹង \(A\) និង \(B\) រៀងគ្នាធៀបនឹង \(O\)។
២. ប្រសិនបើ \(O\) ជាចំណុចកណ្ដាលនៃ \([AB]\) តើចំណុចឆ្លុះនៃ \(A\) ធៀបនឹង \(O\) គឺជាចំណុចណា?

រូបឆ្លុះធៀបនឹងចំណុច និងលក្ខណៈ

លក្ខណៈនៃរូបឆ្លុះធៀបនឹងចំណុច

  • អង្កត់ឆ្លុះ ៖ អង្កត់ពីរឆ្លុះគ្នាធៀបនឹងចំណុចមួយ ជាអង្កត់ស្របគ្នា និងប៉ុនគ្នា (\(AB \parallel A'B'\) និង \(AB = A'B'\))។

  • មុំឆ្លុះ ៖ មុំពីរឆ្លុះគ្នាធៀបនឹងចំណុចមួយ ជាមុំប៉ុនគ្នា (\(\widehat{A'B'C'} = \widehat{ABC}\))។

  • ត្រីកោណឆ្លុះ ៖ ត្រីកោណពីរឆ្លុះគ្នាធៀបនឹងចំណុចមួយ ជាត្រីកោណប៉ុនគ្នា (\(\triangle A'B'C' \cong \triangle ABC\)) និងមានផ្ទៃស្មើគ្នា។

  • ផ្ចិតឆ្លុះនៃរូប ៖ ប្រលេឡូក្រាម ចតុកោណកែង ការ៉េ ចតុកោណស្មើ មានផ្ចិតឆ្លុះត្រង់ចំណុចប្រសព្វអង្កត់ទ្រូង។ រង្វង់មានផ្ចិតឆ្លុះត្រង់ផ្ចិតរបស់វា។

ក. រកប្រវែងជ្រុងទាំងបីនៃ \(\triangle A'B'C'\) ដែលជារូបឆ្លុះនៃ \(\triangle ABC\) ធៀបនឹង \(O\)។ ខ. ប្រៀបធៀបក្រឡាផ្ទៃ និងទិសដៅ \((AB)\) និង \((A'B')\)។

ដំណោះស្រាយ

ដំណោះស្រាយលម្អិត៖
 ដោយភាពឆ្លុះរក្សាប្រវែង ៖ \(A'B'=5\text{ cm}, B'C'=6\text{ cm}, A'C'=4\text{ cm}\)។ ដូចនេះ \(\boxed{(AB) \parallel (A'B')}\) និង \(\boxed{S_{A'B'C'} = S_{ABC}}\)។

១. ចតុកោណកែង \(ABCD\) មួយមានក្រឡាផ្ទៃ \(24\text{ cm}^2\)។ រកក្រឡាផ្ទៃនៃរូបភាពឆ្លុះ \(A'B'C'D'\) របស់វាធៀបនឹងចំណុច \(O\) មួយ។
២. តើត្រីកោណសម័ង្សមានផ្ចិតឆ្លុះដែរឬទេ? ចុះរង្វង់មួយមានផ្ចិតឆ្លុះនៅត្រង់ណា?

ភាពឆ្លុះធៀបនឹងបន្ទាត់ (អ័ក្សឆ្លុះ)

ចំណុចឆ្លុះធៀបនឹងបន្ទាត់មួយ

  • និយមន័យ ៖ ចំណុច \(M'\) ឆ្លុះនឹងចំណុច \(M\) ធៀបនឹងបន្ទាត់ \((\Delta)\) កាលណា \((\Delta)\) ជាមេដ្យាទ័រនៃអង្កត់ \([MM']\)។

  • អ័ក្សឆ្លុះ ៖ បន្ទាត់ \((\Delta)\) ហៅថា អ័ក្សឆ្លុះ នៃភាពឆ្លុះ។

  • លក្ខណៈ ៖ \((\Delta) \perp [MM']\) ត្រង់ចំណុចកណ្ដាល \(H\) នៃ \([MM']\) (\(MH = M'H\))។

  • ចំណុចលើអ័ក្ស ៖ បើ \(M \in (\Delta)\) នោះរូបភាពឆ្លុះរបស់វាគឺខ្លួនវាផ្ទាល់ (\(M' \equiv M\))។

ឧទាហរណ៍

គេឱ្យបន្ទាត់ \((\Delta)\) មួយ និងចំណុច \(A\) នៅក្រៅ \((\Delta)\) ដោយចម្ងាយពី \(A\) ទៅ \((\Delta)\) ស្មើនឹង \(3\text{ cm}\)។
ក. សង់ចំណុច \(A'\) ឆ្លុះនឹង \(A\) ធៀបនឹង \((\Delta)\)។ ខ. គណនា \(AA'\)។ បើ \(B \in (\Delta)\) រកចំណុចឆ្លុះ \(B'\)។

ដំណោះស្រាយ

ដំណោះស្រាយលម្អិត៖
 ទម្លាក់កែងពី \(A\) មក \((\Delta)\) ត្រង់ \(H \implies AH = 3\text{ cm}\)។ ពន្យារ \([AH)\) ឱ្យបាន \(HA' = AH = 3\text{ cm}\)។
 ប្រវែង \(AA' = AH + HA' = 3 + 3 = 6\text{ cm}\) នាំឱ្យ \(\boxed{AA' = 6\text{ cm}}\)។ ដោយ \(B \in (\Delta)\) នោះ \(\boxed{B' \equiv B}\)។

ភាពឆ្លុះធៀបនឹងបន្ទាត់

១. គេមានបន្ទាត់ \((\Delta)\) និងអង្កត់ \([AB]\) ស្របនឹង \((\Delta)\) ដោយមានចម្ងាយ \(2\text{ cm}\) ពី \((\Delta)\)។ តើអង្កត់ឆ្លុះ \([A'B']\) មានប្រវែង និងទីតាំងយ៉ាងដូចម្ដេចធៀបនឹង \([AB]\)?
២. បើ \((\Delta)\) កាត់តាមចំណុច \(A\) តើចំណុចឆ្លុះ \(A'\) ធៀបនឹង \((\Delta)\) នៅត្រង់ណា?

រូបដែលមានអ័ក្សឆ្លុះ

  • និយមន័យ ៖ រូបមួយមានអ័ក្សឆ្លុះ កាលណារូបឆ្លុះរបស់វាធៀបនឹងបន្ទាត់នោះត្រួតស៊ីគ្នាទាំងស្រុងជាមួយរូបដើម។

  • ចំនួនអ័ក្សឆ្លុះនៃរូបធរណីមាត្រ ៖

    • ត្រីកោណសមបាត (\(1\))  ត្រីកោណសម័ង្ស (\(3\))  ចតុកោណកែង (\(2\))

    • ចតុកោណស្មើ (\(2\))  ការ៉េ (\(4\))  រង្វង់ (រាប់មិនអស់ កាត់តាមផ្ចិត)

ឧទាហរណ៍

គេឱ្យរូបចតុកោណកែង \(ABCD\) មួយ។
ក. តើចតុកោណកែងមានអ័ក្សឆ្លុះប៉ុន្មាន? ចូររៀបរាប់ពីអ័ក្សឆ្លុះទាំងនោះ។
ខ. តើបន្ទាត់ផ្ទុកអង្កត់ទ្រូងនៃចតុកោណកែងជាអ័ក្សឆ្លុះរបស់វាដែរឬទេ? ព្រោះអ្វី?

ដំណោះស្រាយ

ដំណោះស្រាយលម្អិត៖
 ចតុកោណកែងមានអ័ក្សឆ្លុះចំនួន \(\boxed{2}\) គឺបន្ទាត់កាត់តាមចំណុចកណ្ដាលនៃជ្រុងឈមទាំងពីរគូ (មេដ្យាទ័រជ្រុងឈម)។
 បន្ទាត់ផ្ទុកអង្កត់ទ្រូងមិនមែនជាអ័ក្សឆ្លុះនៃចតុកោណកែងទេ ពីព្រោះនៅពេលបត់រូបតាមអង្កត់ទ្រូង កំពូលទាំងពីរសងខាងមិនត្រួតស៊ីគ្នាឡើយ (លើកលែងតែករណីការ៉េ)។

១. រកចំនួនអ័ក្សឆ្លុះនៃរូបខាងក្រោម ៖ ក. មុំស្រួចមួយ ខ. ចតុកោណស្មើ គ. ប៉ង់តាកោណនិយ័ត (៥ ជ្រុងស្មើ)។
២. តើត្រីកោណកែងមិនសមបាតមានអ័ក្សឆ្លុះដែរឬទេ?

បំប្លែងកិល (Translation)

បំប្លែងកិល និងលក្ខណៈរូបភាព

  • និយមន័យ ៖ បំប្លែងកិលតាមវ៉ិចទ័រ \(\vec{u}\) គឺជាបំប្លែងដែលរំកិលគ្រប់ចំណុច \(M\) ទៅ \(M'\) ដោយ \(\vec{MM'} = \vec{u}\)។

  • លក្ខណៈ ៖ អង្កត់ភ្ជាប់ចំណុចដើមទៅចំណុចរូបភាពមានទិសស្របគ្នា ទិសដៅដូចគ្នា និងមានប្រវែងស្មើគ្នា \(MM' = u\)។

  • ការរក្សាទំហំ ៖ បំប្លែងកិលរក្សាប្រវែងអង្កត់ (\(AB = A'B'\)), រក្សារង្វាស់មុំ, រក្សាភាពត្រង់ជួរ និងរក្សាក្រឡាផ្ទៃ។

  • រូបភាពនៃរូបធរណីមាត្រ ៖ រូបភាពនៃបន្ទាត់មួយជាបន្ទាត់ស្របនឹងវា, រូបភាពនៃត្រីកោណមួយជាត្រីកោណប៉ុនគ្នាដែលមានទិសដូចគ្នា។

ក. រកកូអរដោនេនៃចំណុចរូបភាព \(A'\)។
ខ. បើអង្កត់ \([AB]\) មានប្រវែង \(5\text{ cm}\) តើអង្កត់រូបភាព \([A'B']\) តាមបំប្លែងកិលនេះមានប្រវែងប៉ុន្មាន?

ដំណោះស្រាយ

ដំណោះស្រាយលម្អិត៖
 រំកិលទៅស្ដាំ ៤ ឯកតា ៖ \(x' = x + 4 = 2 + 4 = 6\)
រំកិលឡើងលើ ២ ឯកតា ៖ \(y' = y + 2 = 3 + 2 = 5 \implies\) \(\boxed{A'(6, 5)}\)
 ដោយបំប្លែងកិលរក្សាប្រវែងអង្កត់ នោះប្រវែងអង្កត់រូបភាពគឺ \(\boxed{A'B' = AB = 5\text{ cm}}\)។

២លំហាត់ទី ២

  1. សង់ចំណុច \(M'\) ឆ្លុះនឹង \(M\) ធៀបនឹង \(O\)

៣លំហាត់ទី ៣

  1. សង់ចំណុចឆ្លុះ \(A'\) នៃ \(A\) ធៀបនឹង \(O\)

  2. គណនាប្រវែង \(A'B\)

  2. រកក្រឡាផ្ទៃនៃរូបភាពចតុកោណកែងថ្មី

  • ផ្ទៃដើម ៖ \(S = 8 \times 5 = 40\text{ cm}^2\)

  • ផ្ទៃថ្មី ៖ \(S' = S = 40\text{ cm}^2\)

៤៦លំហាត់ទី ៤៦

គេមានចំណុច \(M(1, 2)\) និងរូបភាពរបស់វា \(M'(5, 6)\) តាមបំប្លែងកិល ៖

  1. រកចម្ងាយរំកិលតាមអ័ក្សដេក និងអ័ក្សឈរ

  2. រកកូអរដោនេនៃរូបភាព \(N'\) បើ \(N(3, 1)\)

មើលដំណោះស្រាយ

  • ចម្ងាយដេក \(5 - 1 = 4\); ចម្ងាយឈរ \(6 - 2 = 4\)

  • \(N'(3 + 4, 1 + 4) = N'(7, 5)\)

៤៧លំហាត់ទី ៤៧

រង្វង់មួយមានសមីការកាំ \(R = 5\text{ cm}\) ត្រូវបានរំកិលទៅទីតាំងថ្មី ៖

  1. រកកាំនៃរង្វង់រូបភាពថ្មី

  2. គណនាក្រឡាផ្ទៃថាសនៃរង្វង់ថ្មី (យក \(\pi \approx 3.14\))

មើលដំណោះស្រាយ

  • \(R' = R = 5\text{ cm}\)

  • \(S = 3.14 \times 25 = 78.5\text{ cm}^2\)

៤៨លំហាត់ទី ៤៨

គេរំកិលត្រីកោណ \(ABC\) តាមវ៉ិចទ័រ \(\vec{AA'} = \vec{u}\) ៖

  1. បង្ហាញថាចតុកោណ \(ABB'A'\) ជាប្រលេឡូក្រាម

  2. ទាញបញ្ជាក់ថា \(A'B' = AB\)

មើលដំណោះស្រាយ

  • \(\vec{AA'} = \vec{BB'} \Rightarrow AA' \parallel BB'\) និង \(AA' = BB'\) នាំឱ្យ \(ABB'A'\) ជាប្រលេឡូក្រាម។

  • ជ្រុងឈមប្រលេឡូក្រាមស្មើគ្នា នាំឱ្យ \(A'B' = AB\)។

៤៩លំហាត់ទី ៤៩

គេមានចំណុច \(P(-2, 3)\) ត្រូវរំកិលទៅស្ដាំ ៣ ឯកតា និងចុះក្រោម ៥ ឯកតា ៖

  1. រកកូអរដោនេនៃរូបភាព \(P'\)

  2. តើ \(P'\) ស្ថិតនៅក្នុងកាដ្រង់ណា?

មើលដំណោះស្រាយ

  • \(P'(-2 + 3, 3 - 5) = P'(1, -2)\)

  • \(x > 0, y < 0 \Rightarrow P'\) នៅ កាដ្រង់ទី ៤។

៥០លំហាត់ទី ៥០

គេមានបន្ទាត់ \((d)\) មួយត្រូវបានរំកិលតាមវ៉ិចទ័រ \(\vec{u}\) ទៅជាបន្ទាត់ \((d')\) ៖

  1. បញ្ជាក់ពីទីតាំងធៀបរវាង \((d)\) និង \((d')\)

  2. តើនៅពេលណាដែល \((d') \equiv (d)\)?

មើលដំណោះស្រាយ

  • \((d') \parallel (d)\) (ស្របគ្នា)

  • \((d') \equiv (d)\) នៅពេល \(\vec{u}\) ស្របនឹង \((d)\) ឬ \(\vec{u} = \vec{0}\)។
