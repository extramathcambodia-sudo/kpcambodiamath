ប្រវត្តិនៃប្រភាគ និងរបារបន្ទាត់ប្រភាគ

សញ្ញាណនៃប្រភាគបានកកើតឡើងតាំងពីរាប់ពាន់ឆ្នាំមុនគ.ស. នៅពេលដែលមនុស្សត្រូវការបែងចែកដីធ្លី ស្បៀងអាហារ និងវាស់វែងបរិមាណដែលមិនពេញមួយឯកតា។

ក្នុងអារ្យធម៌អេហ្ស៊ីបបុរាណ (ប្រមាណ ១៦៥០ ឆ្នាំមុន គ.ស.) ដូចដែលបានកត់ត្រាក្នុងក្រដាសប៉ាពីរុស Rhind Mathematical Papyrus ជនជាតិអេហ្ស៊ីបបានប្រើប្រាស់តែ «ប្រភាគឯកតា» (Unit Fractions) ដែលមានភាគយកស្មើនឹង \(1\) ប៉ុណ្ណោះ ដូចជា \(\frac{1}{2}, \frac{1}{3}, \frac{1}{4}, \dots\) (លើកលែងតែប្រភាគពិសេស \(\frac{2}{3}\)) ហើយរាល់ប្រភាគផ្សេងៗត្រូវបានបំបែកជាផលបូកនៃប្រភាគឯកតា។ រីឯជនជាតិបាប៊ីឡូនបានប្រើប្រាស់ប្រភាគដែលមានភាគបែងជាប្រព័ន្ធគោល ៦០ សម្រាប់កិច្ចការតារាសាស្ត្រ និងការវាស់មុំ។

នៅប្រទេសឥណ្ឌា គណិតវិទូបុរាណ Aryabhata (សតវត្សទី៥) និង Brahmagupta (សតវត្សទី៧) បានសរសេរប្រភាគដោយដាក់ភាគយកនៅពីលើភាគបែង ប៉ុន្តែពុំទាន់មានរបារបន្ទាត់ខណ្ឌកណ្ដាលឡើយ (ឧទាហរណ៍ សរសេរលេខ \(3\) នៅលើលេខ \(4\))។

លុះមកដល់សតវត្សទី១២ គណិតវិទូអារ៉ាប់ដើមកំណើតម៉ារ៉ុក Abu Bakr al-Hassar បានបង្កើតរបារបន្ទាត់ផ្ដេកខណ្ឌចែករវាងភាគយក និងភាគបែងជាលើកដំបូងក្នុងប្រវត្តិសាស្ត្រ ដែលត្រូវបានគេស្គាល់ជាសកលរហូតដល់សព្វថ្ងៃ បន្ទាប់ពីគណិតវិទូអ៊ីតាលី Leonardo Fibonacci បាននាំយកមកផ្សព្វផ្សាយនៅទ្វីបអឺរ៉ុបតាមរយៈសៀវភៅ Liber Abaci ក្នុងឆ្នាំ ១២០២។

ប្រភាគ

ប្រភាគ គឺជាចំនួនដែលសរសេរក្រោមទម្រង់ \(\dfrac{a}{b}\) ដែល ៖

  • \(a\) ជាចំនួនគត់រ៉ឺឡាទីប ហៅថា ភាគយក (Numerator)។

  • \(b\) ជាចំនួនគត់រ៉ឺឡាទីបខុសពីសូន្យ (\(b \ne 0\)) ហៅថា ភាគបែង (Denominator)។

អត្ថន័យធរណីមាត្រ ៖ ភាគបែង \(b\) បង្ហាញពីចំនួនចំណែកស្មើៗគ្នាដែលឯកតាទាំងមូលត្រូវបានបែងចែក ហើយភាគយក \(a\) បង្ហាញពីចំនួនចំណែកដែលត្រូវបានជ្រើសរើស ឬផាត់ពណ៌។

រូបភាពនេះកំពុងរៀបចំ

ប្រភេទនៃប្រភាគ និងចំនួនចម្រុះ

  • ប្រភាគសុទ្ធ (Proper Fraction) ៖ ប្រភាគដែលភាគយកតូចជាងភាគបែង (\(|a| < |b|\)) ដូចជា \(\frac{2}{3}, \frac{5}{7}, -\frac{3}{8}\) (តម្លៃរបស់វាតូចជាង \(1\))។

  • ប្រភាគមិនសុទ្ធ (Improper Fraction) ៖ ប្រភាគដែលភាគយកធំជាង ឬស្មើភាគបែង (\(|a| \ge |b|\)) ដូចជា \(\frac{7}{4}, \frac{11}{5}, \frac{9}{9}\) (តម្លៃរបស់វាធំជាង ឬស្មើ \(1\))។

  • ចំនួនចម្រុះ (Mixed Number) ៖ ចំនួនដែលផ្សំឡើងដោយចំនួនគត់ និងប្រភាគសុទ្ធ ៖

    \[q\frac{r}{b} = q + \frac{r}{b} = \frac{q \times b + r}{b} \qquad{} (r < b)\]
    ឧទាហរណ៍ ៖ \(\dfrac{11}{4} = \dfrac{2 \times 4 + 3}{4} = 2\dfrac{3}{4}\)។

ឧទាហរណ៍

ក. បំប្លែងប្រភាគមិនសុទ្ធ \(\frac{17}{5}\) និង \(\frac{29}{6}\) ជាចំនួនចម្រុះ។
ខ. បំប្លែងចំនួនចម្រុះ \(3\frac{2}{7}\) និង \(5\frac{3}{4}\) ជាប្រភាគមិនសុទ្ធ។

ដំណោះស្រាយ

ដំណោះស្រាយលម្អិត៖

  1. បំប្លែងជាចំនួនចម្រុះ ៖

    • \(17 \div 5 = 3\) សល់សំណល់ \(2 \implies \dfrac{17}{5} = 3\dfrac{2}{5}\)

    • \(29 \div 6 = 4\) សល់សំណល់ \(5 \implies \dfrac{29}{6} = 4\dfrac{5}{6}\)

  2. បំប្លែងជាប្រភាគមិនសុទ្ធ ៖

    • \(3\dfrac{2}{7} = \dfrac{3 \times 7 + 2}{7} = \dfrac{21 + 2}{7} = \dfrac{23}{7}\)

    • \(5\dfrac{3}{4} = \dfrac{5 \times 4 + 3}{4} = \dfrac{20 + 3}{4} = \dfrac{23}{4}\)

កំណត់ និងបំប្លែងប្រភាគ

១. បំប្លែងប្រភាគមិនសុទ្ធខាងក្រោមជាចំនួនចម្រុះ ៖

  1. \(\dfrac{23}{4}\)

  2. \(\dfrac{37}{7}\)

  3. \(\dfrac{45}{8}\)

២. បំប្លែងចំនួនចម្រុះខាងក្រោមជាប្រភាគមិនសុទ្ធ ៖

  1. \(4\dfrac{3}{5}\)

  2. \(6\dfrac{1}{8}\)

  3. \(2\dfrac{7}{9}\)

ប្រភាគសមមូល

ពីរប្រភាគ \(\dfrac{a}{b}\) និង \(\dfrac{c}{d}\) ហៅថាជាប្រភាគសមមូល ឬស្មើគ្នា (កំណត់សរសេរ \(\dfrac{a}{b} = \dfrac{c}{d}\)) លុះត្រាតែផលគុណខ្វែងរបស់វាស្មើគ្នា ៖

\[\frac{a}{b} = \frac{c}{d} \iff a \times d = b \times c \qquad{} (b \ne 0, d \ne 0)\]

លក្ខណៈគ្រឹះនៃប្រភាគសមមូល

បើគេគុណ ឬចែកភាគយក និងភាគបែងនៃប្រភាគមួយនឹងចំនួនគត់ខុសពីសូន្យ \(k\) តែមួយ នោះគេទទួលបានប្រភាគថ្មីមួយសមមូលនឹងប្រភាគដើម ៖

\[\frac{a}{b} = \frac{a \times k}{b \times k} \qquad{} \text{និង} \qquad{} \frac{a}{b} = \frac{a \div k}{b \div k} \qquad{} (k \ne 0)\]

ឧទាហរណ៍

ក. រកតម្លៃ \(x\) បើគេដឹងថា \(\dfrac{x}{5} = \dfrac{24}{30}\)។
ខ. បំពេញចំនួនក្នុងប្រអប់ \(\Box\) ៖ \(\dfrac{3}{7} = \dfrac{\Box}{28} = \dfrac{27}{\Box}\)។

ដំណោះស្រាយ

ដំណោះស្រាយលម្អិត៖

  1. ដោយ \(\dfrac{x}{5} = \dfrac{24}{30} \implies x \times 30 = 5 \times 24 \implies 30x = 120 \implies x = \dfrac{120}{30} = 4\)។

  2. គេមាន \(\dfrac{3}{7} = \dfrac{3 \times 4}{7 \times 4} = \dfrac{12}{28} \implies \Box = 12\)។
    និង \(\dfrac{3}{7} = \dfrac{3 \times 9}{7 \times 9} = \dfrac{27}{63} \implies \Box = 63\)។

ស្វែងរកតម្លៃមិនស្គាល់ក្នុងប្រភាគសមមូល

១. រកតម្លៃ \(x\) ក្នុងសមភាពខាងក្រោម ៖

  1. \(\dfrac{x}{8} = \dfrac{21}{56}\)

  2. \(\dfrac{5}{12} = \dfrac{x}{60}\)

  3. \(\dfrac{x - 1}{4} = \dfrac{6}{8}\)

  4. \(\dfrac{3}{x} = \dfrac{-15}{25}\)

២. បំពេញប្រអប់ ៖ \(\dfrac{2}{3} = \dfrac{\Box}{12} = \dfrac{10}{\Box} = \dfrac{\Box}{30} = \dfrac{24}{\Box}\)។

ការសម្រួលប្រភាគ

សម្រួលប្រភាគ គឺការចែកភាគយក និងភាគបែងនៃប្រភាគនឹងតួចែករួមធំបំផុត (\(\gcd\)) របស់វា ដើម្បីទទួលបានប្រភាគបង្រួមមិនបាន (Irreducible Fraction) ដែលសមមូលនឹងប្រភាគដើម។

  • ប្រភាគ \(\dfrac{a}{b}\) ជាប្រភាគបង្រួមមិនបាន លុះត្រាតែ \(\gcd(|a|, |b|) = 1\)។

  • សញ្ញានៃប្រភាគ ៖ \(\dfrac{-a}{b} = \dfrac{a}{-b} = -\dfrac{a}{b}\) និង \(\dfrac{-a}{-b} = \dfrac{a}{b}\)។

ឧទាហរណ៍

សម្រួលប្រភាគខាងក្រោមឱ្យទៅជាប្រភាគបង្រួមមិនបាន ៖

  1. \(\dfrac{48}{72}\)

  2. \(\dfrac{-35}{105}\)

  3. \(\dfrac{-84}{-126}\)

ដំណោះស្រាយ

ដំណោះស្រាយលម្អិត៖

  1. \(\gcd(48, 72) = 24 \implies \dfrac{48}{72} = \dfrac{48 \div 24}{72 \div 24} = \dfrac{2}{3}\)

  2. \(\gcd(35, 105) = 35 \implies \dfrac{-35}{105} = \dfrac{-35 \div 35}{105 \div 35} = -\dfrac{1}{3}\)

វិធាននៃការប្រៀបធៀបប្រភាគ

  • ប្រភាគមានភាគបែងដូចគ្នា ៖ ប្រភាគណាដែលមានភាគយកធំជាង ជាប្រភាគធំជាង ៖

    \[\frac{a}{m} > \frac{b}{m} \iff a > b \qquad{} (m > 0)\]

  • ប្រភាគមានភាគយកដូចគ្នា (\(a > 0\)) ៖ ប្រភាគណាដែលមានភាគបែងតូចជាង ជាប្រភាគធំជាង ៖

    \[\frac{a}{b} > \frac{a}{c} \iff b < c \qquad{} (b, c > 0)\]

  • ប្រភាគមានភាគបែងខុសគ្នា ៖ គេត្រូវតម្រូវភាគបែងរួម (យកពហុគុណរួមតូចបំផុត \(\text{lcm}\) នៃភាគបែងធ្វើជាភាគបែងរួម) រួចប្រៀបធៀបភាគយក និងភាគយក។

វិធីបូក និងវិធីដកប្រភាគ

វិធាននៃវិធីបូក និងវិធីដកប្រភាគ

  • ប្រភាគមានភាគបែងដូចគ្នា ៖ គេបូក ឬដកតែភាគយក និងភាគយក ហើយរក្សាភាគបែងរួមដដែល ៖

    \[\frac{a}{c} + \frac{b}{c} = \frac{a + b}{c}, \qquad{} \frac{a}{c} - \frac{b}{c} = \frac{a - b}{c} \qquad{} (c \ne 0)\]

  • ប្រភាគមានភាគបែងខុសគ្នា ៖ គេត្រូវតម្រូវភាគបែងរួម (តាមរយៈ \(\text{lcm}\)) រួចបូក ឬដកភាគយក ៖

    \[\frac{a}{b} \pm \frac{c}{d} = \frac{a \times d \pm b \times c}{b \times d} \qquad{} (b, d \ne 0)\]

វិធីគុណ និងវិធីចែកប្រភាគ

វិធាននៃវិធីគុណ និងវិធីចែកប្រភាគ

  • វិធីគុណ ៖ គេគុណភាគយកនឹងភាគយក ហើយគុណភាគបែងនឹងភាគបែង ៖

    \[\frac{a}{b} \times \frac{c}{d} = \frac{a \times c}{b \times d} \qquad{} (b \ne 0, d \ne 0)\]

  • ចម្រាសនៃប្រភាគ ៖ ចម្រាសនៃប្រភាគ \(\dfrac{a}{b}\) (\(a \ne 0, b \ne 0\)) គឺ \(\dfrac{b}{a}\) ពីព្រោះ \(\dfrac{a}{b} \times \dfrac{b}{a} = 1\)។

  • វិធីចែក ៖ ដើម្បីចែកប្រភាគមួយនឹងប្រភាគមួយទៀត គេត្រូវយកប្រភាគទីមួយគុណនឹងចម្រាសនៃប្រភាគទីពីរ ៖

    \[\frac{a}{b} \div \frac{c}{d} = \frac{a}{b} \times \frac{d}{c} = \frac{a \times d}{b \times c} \qquad{} (b, c, d \ne 0)\]

    \(\dfrac{3}{8} + \dfrac{2}{5} = \dfrac{15 + 16}{40} = \dfrac{31}{40}\) នៃផ្លូវ

  • ថ្ងៃទីបីនៅសល់ប្រភាគ ៖
    \(1 - \dfrac{31}{40} = \dfrac{9}{40}\) នៃផ្លូវ ដែលត្រូវនឹង \(45\text{ km}\)

  • ប្រវែងផ្លូវសរុប ៖
    \(45 \div \dfrac{9}{40} = 45 \times \dfrac{40}{9} = 200\text{ km}\)

៤៧លំហាត់ទី ៤៧

ដីឡូត៍មួយមានបណ្ដោយ \(\dfrac{35}{2}\text{ m}\) និងទទឹង \(\dfrac{24}{5}\text{ m}\)។

  1. គណនាបរិមាណក្រឡាផ្ទៃដីឡូត៍នោះ។

  2. គណនាបរិមាត្រនៃដីឡូត៍។

មើលដំណោះស្រាយ

គណនាក្រឡាផ្ទៃ និងបរិមាត្រដីឡូត៍ ៖

  • ក្រឡាផ្ទៃ ៖
    \(S = \dfrac{35}{2} \times \dfrac{24}{5} = \dfrac{35 \times 24}{10} = 84\text{ m}^2\)

  • បរិមាត្រ ៖
    \(P = 2 \times \left(\dfrac{35}{2} + \dfrac{24}{5}\right) = 2 \times \dfrac{175 + 48}{10} = \dfrac{223}{5} = 44.6\text{ m}\)

៤៨លំហាត់ទី ៤៨

ធុងប្រេងមួយមានប្រេង \(\dfrac{3}{4}\) នៃចំណុះធុង។ បន្ទាប់ពីគេដងប្រេងចេញអស់ \(30\) លីត្រ ប្រេងក្នុងធុងនៅសល់ \(\dfrac{2}{5}\) នៃចំណុះធុង។ រកចំណុះសរុបនៃធុងប្រេងនោះ។

មើលដំណោះស្រាយ

រកចំណុះសរុបនៃធុងប្រេង ៖

  • ប្រភាគប្រេងដែលដងចេញ \(30\text{ L}\) គឺ ៖
    \(\dfrac{3}{4} - \dfrac{2}{5} = \dfrac{15 - 8}{20} = \dfrac{7}{20}\) នៃចំណុះធុង

  • ចំណុះសរុបនៃធុង ៖
    \(30 \div \dfrac{7}{20} = \dfrac{600}{7} \approx 85.71\) លីត្រ (ឬ \(85\dfrac{5}{7}\text{ L}\))

៤៩លំហាត់ទី ៤៩

ឪពុកម្នាក់បានចែកទ្រព្យសម្បត្តិឱ្យកូន ៣ នាក់ ៖ កូនច្បងទទួលបាន \(\dfrac{3}{7}\) នៃទ្រព្យ កូនកណ្ដាលទទួលបាន \(\dfrac{1}{3}\) នៃទ្រព្យ និងកូនពៅទទួលបានប្រាក់ \(20\,000\,000\) រៀល ។ គណនាតម្លៃទ្រព្យសម្បត្តិសរុបរបស់ឪពុក និងប្រាក់ដែលកូនម្នាក់ៗទទួលបាន។

មើលដំណោះស្រាយ

គណនាទ្រព្យសម្បត្តិសរុប និងប្រាក់កូនម្នាក់ៗ ៖

  • កូនច្បង និងកូនកណ្ដាលបាន ៖
    \(\dfrac{3}{7} + \dfrac{1}{3} = \dfrac{9 + 7}{21} = \dfrac{16}{21}\) នៃទ្រព្យ

  • កូនពៅទទួលបាន ៖
    \(1 - \dfrac{16}{21} = \dfrac{5}{21}\) ត្រូវនឹង \(20\,000\,000\) រៀល

  • ទ្រព្យសម្បត្តិសរុប ៖
    \(20\,000\,000 \div \dfrac{5}{21} = 84\,000\,000\) រៀល

  • កូនច្បង ៖ \(84\,000\,000 \times \dfrac{3}{7} = 36\,000\,000\) រៀល

  • កូនកណ្ដាល ៖ \(84\,000\,000 \times \dfrac{1}{3} = 28\,000\,000\) រៀល

៥០លំហាត់ទី ៥០

កម្រងប្រភាគមួយមានតួ ៖ \(\dfrac{1}{2}, \dfrac{2}{3}, \dfrac{3}{4}, \dfrac{4}{5}, \dots\)។

  1. ចូរសរសេររូបមន្តតួទូទៅ \(u_n\) នៃកម្រងប្រភាគនេះ។

  2. តើតួទី ១០០ នៃកម្រងនេះស្មើនឹងប៉ុន្មាន?

មើលដំណោះស្រាយ

កម្រងប្រភាគ \(\dfrac{1}{2}, \dfrac{2}{3}, \dfrac{3}{4}, \dots\) ៖

  • តួទូទៅ ៖ \(u_n = \dfrac{n}{n+1}\)

  • តួទី ១០០ គឺ ៖ \(u_{100} = \dfrac{100}{100+1} = \dfrac{100}{101}\)
